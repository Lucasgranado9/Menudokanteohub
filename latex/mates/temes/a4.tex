\subsection*{Anàlisi}

\begin{itemize}
\item \textbf{Recta tangent} a $y=f(x)$ en $x=a$: $y-f(a)=f'(a)(x-a)$. \textbf{Recta normal:} $y-f(a)=-\dfrac{1}{f'(a)}(x-a)$.
\item \textbf{Límits notables:} $\displaystyle\lim_{x\to0}\frac{\sin x}{x}=1$, $\ \lim_{x\to0}\frac{e^x-1}{x}=1$, $\ \lim_{x\to\infty}\left(1+\frac1x\right)^x=e$.
\item \textbf{Indeterminació $1^\infty$:} $\displaystyle\lim f^g=e^{\lim g\,(f-1)}$.
\item \textbf{Teorema fonamental:} $\dfrac{d}{dx}\displaystyle\int_a^{u(x)}f(t)\,dt=f(u(x))\,u'(x)$.
\item \textbf{Àrea entre corbes:} $S=\displaystyle\int_a^b(f_{\text{sup}}-f_{\text{inf}})\,dx$ (dividint en trams si es creuen).
\end{itemize}

\subsection*{Àlgebra}

\begin{itemize}
\item \textbf{Equacions matricials:} $AX=B\Rightarrow X=A^{-1}B$; $\ XA=B\Rightarrow X=BA^{-1}$; $\ AX+X=B\Rightarrow X=(A+I)^{-1}B$.
\item \textbf{Inversa per Gauss-Jordan:} $(A\,|\,I)\to(I\,|\,A^{-1})$.
\end{itemize}

\subsection*{Geometria}

\begin{itemize}
\item \textbf{Simètric d'un punt:} $P'=2M-P$, amb $M$ la projecció ortogonal de $P$.
\item \textbf{Pla per tres punts:} $\vec n=\overrightarrow{AB}\times\overrightarrow{AC}$.
\end{itemize}

\subsection*{Probabilitat}

\begin{itemize}
\item $P(A\cup B)=P(A)+P(B)-P(A\cap B)$; $\ P(A\cap B)=P(A)P(B\mid A)$.
\item \textbf{Probabilitat total:} $P(B)=\sum P(A_i)P(B\mid A_i)$; \textbf{Bayes:} $P(A_i\mid B)=\dfrac{P(A_i)P(B\mid A_i)}{P(B)}$.
\item \textbf{Binomial:} $E=np$, $\sigma^2=npq$. \textbf{Tipificació:} $Z=\dfrac{X-\mu}{\sigma}$.
\end{itemize}
