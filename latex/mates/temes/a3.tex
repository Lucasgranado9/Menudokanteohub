\subsection*{Tècniques d'integració}

\begin{itemize}
\item \textbf{Per parts:} $\displaystyle\int u\,dv = uv - \int v\,du$ (prioritat LIATE per triar $u$).
\item \textbf{Per canvi de variable:} $x = u(t) \Rightarrow dx = u'(t)\,dt$, i $\displaystyle\int f(x)\,dx = \int f(u(t))\,u'(t)\,dt$.
\item \textbf{Fraccions parcials:} si $\deg P < \deg Q$, descompondre $\dfrac{P(x)}{Q(x)}$ en fraccions simples.
\end{itemize}
